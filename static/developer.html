const API_KEY = "inv_demo_free_2026";
const BASE    = "https://invisync.vercel.app";

// Encode: generate QR + hide a secret inside it
async function encodeQR(url, secret, password) {
  const fd = new FormData();
  fd.append("url",     url);
  fd.append("message", secret);
  fd.append("password", password);

  const res = await fetch(`${BASE}/api/qr/encode`, {
    method:  "POST",
    headers: { "X-API-Key": API_KEY },
    body: fd,
  });

  console.log("Tokens used:", res.headers.get("X-Tokens-Used"));
  return res.blob();   // PNG blob — still scans to original URL
}

// Encode text with zero-width Unicode
async function hideInText(coverText, secret) {
  const fd = new FormData();
  fd.append("cover",   coverText);
  fd.append("message", secret);

  const res = await fetch(`${BASE}/api/text/encode`, {
    method:  "POST",
    headers: { "X-API-Key": API_KEY },
    body: fd,
  });
  const j = await res.json();
  return j.stego_text;   // looks identical, invisible chars inside
}
