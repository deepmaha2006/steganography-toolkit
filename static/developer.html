Quick Start

From zero to your first encoded image in under 2 minutes.

01

Get a Key

Use inv_demo_free_2026 (100 tokens/day free) or buy a plan above — key arrives instantly after payment.

02

Send a Request

POST to any /api/* endpoint with X-API-Key header and multipart/form-data body.

03

Get the Result

Receive stego file (image/audio/PDF) or JSON. Check X-Tokens-Remaining in response headers.

# Encode a secret message into a PNG image (AES-256-GCM encrypted)
curl -X POST https://invisync.vercel.app/api/image/encode \
  -H "X-API-Key: inv_demo_free_2026" \
  -F "cover=@photo.png" \
  -F "message=Operation Nightfall begins at dawn" \
  -F "password=s3cur3P@ss" \
  --output stego.png

# X-Tokens-Used: 10  |  X-Tokens-Remaining: 90  |  X-Plan: free

# Decode it back
curl -X POST https://invisync.vercel.app/api/image/decode \
  -H "X-API-Key: inv_demo_free_2026" \
  -F "stego=@stego.png" \
  -F "password=s3cur3P@ss"
# → {"message": "Operation Nightfall begins at dawn", "size_bytes": 42}
import requests

API_KEY = "inv_live_your_key_here"
BASE    = "https://invisync.vercel.app"

# Encode — image
with open("photo.png", "rb") as f:
    r = requests.post(
        f"{BASE}/api/image/encode",
        headers={"X-API-Key": API_KEY},
        files={"cover": ("photo.png", f, "image/png")},
        data={"message": "secret text", "password": "pass"},
    )
print("Remaining:", r.headers["X-Tokens-Remaining"])
with open("stego.png", "wb") as out:
    out.write(r.content)

# Decode — QR code

    r2 = requests.post(
        f"{BASE}/api/qr/decode",
        headers={"X-API-Key": API_KEY},
        files={"stego": f},
        data={"password": "pass"},
    )
print(r2.json()["message"])
const API_KEY = "inv_live_your_key_here";
const BASE    = "https://invisync.vercel.app";

// Encode: generate QR + hide secret inside pixel LSBs
async function hideInQR(url, secret, password) {
  const fd = new FormData();
  fd.append("url",      url);
  fd.append("message",  secret);
  fd.append("password", password);
  const res = await fetch(`${BASE}/api/qr/encode`, {
    method: "POST", headers: {"X-API-Key": API_KEY}, body: fd
  });
  console.log("Remaining:", res.headers.get("X-Tokens-Remaining"));
  return res.blob(); // PNG — still scans to original URL
}

// Hide message in text (zero-width Unicode)
async function hideInText(coverText, secret) {
  const fd = new FormData();
  fd.append("cover",   coverText);
  fd.append("message", secret);
  const res = await fetch(`${BASE}/api/text/encode`, {
    method: "POST", headers: {"X-API-Key": API_KEY}, body: fd
  });
  return (await res.json()).stego_text; // looks identical, invisible chars inside
}
